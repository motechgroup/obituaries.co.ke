    function initiatePayment() {

    let currentStep = 1;
    const totalSteps = 4;
    const form = document.getElementById('obituaryForm');
    const nextBtn = document.getElementById('nextBtn');
    const prevBtn = document.getElementById('prevBtn');
    const payButton = document.getElementById('payButton');

    function updateStep() {
      // Hide all panels
      document.querySelectorAll('.form-step-panel').forEach(p => {
        p.classList.add('hidden');
        p.classList.remove('opacity-100');
        p.classList.add('opacity-0');
      });

      // Show current panel
      const currentPanel = document.getElementById(`step-content-${currentStep}`);
      currentPanel.classList.remove('hidden');
      setTimeout(() => {
        currentPanel.classList.remove('opacity-0');
        currentPanel.classList.add('opacity-100');
      }, 50);

      // Update Stepper UI
      document.querySelectorAll('.step-indicator').forEach(indicator => {
        const stepNum = parseInt(indicator.dataset.step);
        if(stepNum <= currentStep) {
          indicator.classList.remove('opacity-40');
          const circle = indicator.querySelector('div');
          if(stepNum < currentStep) {
            circle.classList.replace('bg-primary', 'bg-secondary');
            circle.innerHTML = '<span class="material-symbols-outlined text-[20px]">check</span>';
          } else {
            circle.classList.replace('bg-surface-container-highest', 'bg-primary');
            circle.classList.remove('bg-secondary');
            circle.classList.add('text-on-primary');
          }
        } else {
          indicator.classList.add('opacity-40');
        }
      });

      // Update Nav Buttons
      prevBtn.style.visibility = currentStep === 1 ? 'hidden' : 'visible';
      
      if(currentStep === totalSteps) {
        nextBtn.classList.add('hidden');
      } else {
        nextBtn.classList.remove('hidden');
      }

      window.scrollTo({ top: 0, behavior: 'smooth' });
    }

    nextBtn.addEventListener('click', () => {
      if(currentStep < totalSteps) {
        currentStep++;
        updateStep();
      }
    });

    prevBtn.addEventListener('click', () => {
      if(currentStep > 1) {
        currentStep--;
        updateStep();
      }
    });

    payButton.addEventListener('click', () => {
      payButton.innerHTML = '<span class="animate-spin material-symbols-outlined">refresh</span> Connecting to Safaricom...';
      payButton.disabled = true;
      
      setTimeout(() => {
        const overlay = document.getElementById('successOverlay');

        overlay.classList.add('opacity-100');
        setTimeout(() => icon.classList.add('scale-100'), 200);
      }, 2500);

<script>
    // Subtle Counter Animation for Stats
    function animateValue(id, start, end, duration) {
        if (start === end) return;
        const range = end - start;
        let current = start;
        const increment = end > start? 1 : -1;
        const stepTime = Math.abs(Math.floor(duration / range));
        const obj = document.getElementById(id);
        if(!obj) return;
        
        const timer = setInterval(function() {
            current += increment;
            obj.innerHTML = (id === 'stat-revenue') ? (current / 10).toFixed(1) + 'M' : current.toLocaleString();
            if (current == end) {
                clearInterval(timer);
            }
        }, stepTime);
    }

    // Trigger animations on load
    window.addEventListener('DOMContentLoaded', () => {
        // Just illustrative, assuming values are parsed from the DOM
        animateValue('stat-pending', 0, 42, 1000);
        animateValue('stat-total', 1000, 1284, 1500);
    });

    // Mock action handling
    document.querySelectorAll('button[title]').forEach(btn => {
        btn.addEventListener('click', function() {
            const action = this.getAttribute('title');
            const row = this.closest('tr');
            const name = row.querySelector('.text-primary').innerText;
            
            if (action === 'Approve') {
                if(confirm(`Are you sure you want to approve the obituary for ${name}?`)) {
                    row.style.opacity = '0';
                    row.style.transform = 'translateX(20px)';
                    setTimeout(() => row.remove(), 300);
                }
            } else if (action === 'Reject') {
                 const reason = prompt(`Reason for rejecting ${name}:`);
                 if(reason) {
                    row.style.backgroundColor = 'rgba(186, 26, 26, 0.05)';
                    setTimeout(() => {
                        row.style.opacity = '0';
                        setTimeout(() => row.remove(), 300);
                    }, 500);
                 }
            }
        });
    });
